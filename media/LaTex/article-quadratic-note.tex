\section{Introduction}\label{sec:intro}
A quadratic equation $ax^2 + bx + c = 0$ with $a \neq 0$ has its roots in
$\R$ or in $\mathbb{C}$. They are given by equation~\eqref{eq:roots} in
Section~\ref{sec:formula}; the vector form $\vect{x}$ uses a document
macro.\footnote{Footnotes are collected at the end of the imported

\section{The formula}\label{sec:formula}
\begin{equation}\label{eq:roots}
  x = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a}
\end{equation}
Completing the square gives, in aligned steps,
\begin{align}
  a x^2 + b x + c &= 0 \label{eq:start} \\
  \left(x + \frac{b}{2a}\right)^2 &= \frac{b^2 - 4ac}{4a^2}
\end{align}
and the vertex form $y = a(x - x_v)^2 + y_v$ follows from \eqref{eq:start}.
\[

\begin{theorem}[Vieta]
If $x_1$ and $x_2$ are the roots, then $x_1 + x_2 = -\frac{b}{a}$ and
$x_1 x_2 = \frac{c}{a}$.
\end{theorem}

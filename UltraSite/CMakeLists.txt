# UltraSite/CMakeLists.txt
# UltraSite - the HTML + JavaScript output of a website (UltraPress and the
# UltraStore web store; Docs/Modules/UltraSite/README.md). A headless static
# library: routes in, a folder of pages, a hashed stylesheet and hashed
# script modules out. The stylesheet and the scripts under resources/ are
# compiled into the library, so it needs no file beside it at run time.
# Version: 0.4.0 - the preview server, Brotli (optional) for the .br copies
# Version: 0.2.0 - the Markdown docs, the search, toc, tabs and copy modules
# Version: 0.1.0
# Last Modified: 2026-11-17
# Author: UltraCanvas Framework / ULTRA OS

# resources/<file> -> a byte array in generated/UltraSiteResources.h. Done
# at configure time; CMAKE_CONFIGURE_DEPENDS re-runs the configure when a
# resource changes, so an edited script reaches the next build.
set(_ULTRASITE_RESOURCES
    base.css   kBaseCss
    core.js    kCoreJs
    search.js  kSearchJs
    toc.js     kTocJs
    tabs.js    kTabsJs
    copy.js    kCopyJs
)
set(_ULTRASITE_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated/UltraSiteResources.h")
set(_ULTRASITE_HEADER "// Generated by UltraSite/CMakeLists.txt from UltraSite/resources/ - do not edit.\n#pragma once\n#include <cstddef>\nnamespace UltraSite::Resources {\n")
list(LENGTH _ULTRASITE_RESOURCES _ULTRASITE_COUNT)
math(EXPR _ULTRASITE_LAST "${_ULTRASITE_COUNT} - 1")
foreach(_index RANGE 0 ${_ULTRASITE_LAST} 2)
    math(EXPR _name_index "${_index} + 1")
    list(GET _ULTRASITE_RESOURCES ${_index} _file)
    list(GET _ULTRASITE_RESOURCES ${_name_index} _symbol)
    set(_path "${CMAKE_CURRENT_SOURCE_DIR}/resources/${_file}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_path}")
    file(READ "${_path}" _hex HEX)
    string(LENGTH "${_hex}" _hex_length)
    math(EXPR _size "${_hex_length} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
    string(APPEND _ULTRASITE_HEADER
        "inline constexpr unsigned char ${_symbol}[] = {${_bytes}0x00};\n"
        "inline constexpr std::size_t ${_symbol}Size = ${_size};\n")
endforeach()
string(APPEND _ULTRASITE_HEADER "} // namespace UltraSite::Resources\n")
file(WRITE "${_ULTRASITE_GENERATED}.tmp" "${_ULTRASITE_HEADER}")
configure_file("${_ULTRASITE_GENERATED}.tmp" "${_ULTRASITE_GENERATED}" COPYONLY)   # rewritten only when it changed

add_library(UltraSite STATIC
    core/UltraSiteBuilder.cpp
    core/UltraSiteAssets.cpp
    core/UltraSiteMarkdown.cpp
    core/UltraSiteDocs.cpp
    core/UltraSiteServer.cpp
)
set_target_properties(UltraSite PROPERTIES
    POSITION_INDEPENDENT_CODE ON
    OUTPUT_NAME "ultrasite")
target_compile_features(UltraSite PUBLIC cxx_std_20)
target_include_directories(UltraSite PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/../UltraCanvas/include)
target_include_directories(UltraSite PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/core
    ${CMAKE_CURRENT_BINARY_DIR}/generated)
# UltraCanvasJSON, the HTMLReader (EscapeText, the parser the link check
# reads pages with) and UltraCrypt (SHA-256) live in the core library.
target_link_libraries(UltraSite PUBLIC ${ULTRACANVAS_LIBRARY})

# zlib measures what a page weighs over the wire. Without it the budgets
# compare raw sizes, which only makes them stricter.
find_package(ZLIB QUIET)
if(ZLIB_FOUND)
    target_link_libraries(UltraSite PRIVATE ZLIB::ZLIB)
    target_compile_definitions(UltraSite PRIVATE ULTRASITE_HAS_ZLIB=1)
endif()

# Brotli writes the .br copies BuildOptions::precompress asks for; without
# it the .gz copies are written alone. Optional, as in VirtualFS.
find_package(PkgConfig QUIET)
if(PKG_CONFIG_FOUND)
    pkg_check_modules(ULTRASITE_BROTLI QUIET IMPORTED_TARGET libbrotlienc)
endif()
if(ULTRASITE_BROTLI_FOUND)
    target_link_libraries(UltraSite PRIVATE PkgConfig::ULTRASITE_BROTLI)
    target_compile_definitions(UltraSite PRIVATE ULTRASITE_HAS_BROTLI=1)
endif()

# The preview server (core/UltraSiteServer.cpp) listens through UltraNet,
# which lives in the core library or, under a static core, beside it.
if(TARGET UltraNet)
    target_link_libraries(UltraSite PUBLIC UltraNet)
endif()

message(STATUS "  [+] UltraSite - the HTML + JavaScript site builder")
